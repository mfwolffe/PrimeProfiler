#
# template for makefiles in proj-root/makefiles to build the different C binaries.
# this is only to be used in script to populate the actual makefiles with proper
# compiler optimizations & other flags, targets, etc.
#

CC = gcc
CFLAGS = -Wall -Wextra -O{CANARY}

SRCS = src/main.c src/naive.c src/less_naive.c
HDRS = src/primes.h

OBJS = $(SRCS:.c=.o)
TARGET = build/opt{CANARY}/main
all: $(TARGET)

$(TARGET): $(OBJS)
    $(CC) $(CFLAGS) -o $(TARGET) $(OBJS)

%.o: %.c $(HDRS)
    $(CC) $(CFLAGS) -c $< -o $@

clean:
    rm -f $(OBJS) $(TARGET)
